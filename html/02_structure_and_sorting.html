The model - RCCircuit -- electrical.tiny

// ---------------------------------------------------------------------------
// A minimal electrical component library, in the style of Modelica.Electrical
// ---------------------------------------------------------------------------

connector Pin
  Real v        "electrical potential [V]";
  flow Real i   "current INTO the component [A]";
end Pin;

partial model OnePort "component with two pins, same current in and out"
  Pin p, n;
  Real v "voltage drop p.v - n.v [V]";
  Real i "current from p to n [A]";
equation
  v = p.v - n.v;
  i = p.i;
  p.i + n.i = 0;
end OnePort;

model Resistor
  extends OnePort;
  parameter Real R = 100 "resistance [Ohm]";
equation
  v = R * i;
end Resistor;

model Capacitor
  extends OnePort;
  parameter Real C = 1e-3 "capacitance [F]";
equation
  C * der(v) = i;
end Capacitor;

model Inductor
  extends OnePort;
  parameter Real L = 1e-3 "inductance [H]";
equation
  L * der(i) = v;
end Inductor;

model ConstantVoltage
  extends OnePort;
  parameter Real V = 10 "source voltage [V]";
equation
  v = V;
end ConstantVoltage;

model Ground
  Pin p;
equation
  p.v = 0;
end Ground;

// ---------------------------------------------------------------------------
// A charging RC circuit: the "hello world" of acausal modeling
// ---------------------------------------------------------------------------
model RCCircuit
  ConstantVoltage src(V = 10);
  Resistor        r(R = 100);
  Capacitor       c(C = 1e-3, v(start = 0));
  Ground          gnd;
equation
  connect(src.p, r.p);
  connect(r.n, c.p);
  connect(c.n, src.n);
  connect(src.n, gnd.p);
end RCCircuit;

How this system is actually solved

The blocks above are the plan. This is the procedure that follows from it: what happens to each block, what happens before the first step, and what happens when an event interrupts.

1. Once, before the first step: initialization

Every state simply takes its start value; no equations have to be solved yet:

c.v = 0

2. At every evaluation: the continuous system

The integrator supplies the states c.v, the parameters, time, and asks what the derivatives are. Everything else is unknown and has to be computed, in the order the sorting found:

  1. r.v := src.V - c.vrearranged directly from r.v = src.V - c.v
  2. c.i := r.v/r.Rrearranged directly from r.v = r.R * c.i
  3. der(c.v) := c.i/c.Crearranged directly from c.C * der(c.v) = c.i
  4. gnd.p.i := 0rearranged directly from c.i - c.i + gnd.p.i = 0

The derivatives der(c.v) go back to the integrator. Every other value computed on the way is kept as a result, and the variables that alias elimination removed are recovered from the ones that survived.

3. In between: the integrator

Between evaluations the integrator advances the states -- either choosing its own step size to meet a tolerance, or taking the fixed step you gave it. That is the only part of this whole pipeline that TinySim does not do itself.

The generated simulation model

This is the code that actually runs. It is generated from the sorted blocks above, one block at a time.

# ==========================================================================
# Simulation code generated by TinySim for model 'RCCircuit'.
#
# 4 equations, 4 unknowns, sorted into 4 blocks (no algebraic loops)
# States: c.v
#
# Dots in model names become double underscores here: c.v -> c__v.
# ==========================================================================

from math import (sin, cos, tan, asin, acos, atan, atan2, exp, log,
                  log10, sqrt, tanh, fabs as abs)
import numpy as np
from scipy.optimize import fsolve


def _solve_block(residual, guess, block, unknowns, t):
    """
    Solve one algebraic block numerically.

    An iterative solver needs a starting point, and a bad one can
    make it fail on a system that has a perfectly good solution --
    which is exactly why `start` values matter for the variables of
    an algebraic loop.  This tries the remembered guess first, then
    a fresh start from zero, and refuses to continue quietly if
    neither works.
    """
    for attempt in (guess, [0.0] * len(guess)):
        values, _, flag, message = fsolve(residual, attempt,
                                          full_output=True)
        if flag == 1:
            return values
    raise RuntimeError(
        'block %d (%s) did not converge at t = %g: %s'
        % (block, unknowns, t, message.strip()))


def evaluate(t, x, p, d, guess=None):
    """
    Solve the simulation equations of model 'RCCircuit'.

    t     : time
    x     : state vector, in the order listed below
    p     : dict of parameter values, keyed by model name
    d     : dict of discrete-variable values
    guess : dict used to remember iteration start values
    """
    guess = {} if guess is None else guess

    # ---- states: supplied by the integrator ----
    c__v = x[0]    # c.v -- voltage drop p.v - n.v [V]

    # ---- parameters: constant during simulation ----
    c__C = p['c.C']    # c.C -- capacitance [F]
    r__R = p['r.R']    # r.R -- resistance [Ohm]
    src__V = p['src.V']    # src.V -- source voltage [V]

    # ---- block 1: solve for r.v  [explicit]
    #        r.v = src.V - c.v
    r__v = -c__v + src__V

    # ---- block 2: solve for c.i  [explicit]
    #        r.v = r.R * c.i
    c__i = r__v/r__R

    # ---- block 3: solve for der(c.v)  [explicit]
    #        c.C * der(c.v) = c.i
    der_c__v = c__i/c__C

    # ---- block 4: solve for gnd.p.i  [explicit]
    #        c.i - c.i + gnd.p.i = 0
    gnd__p__i = 0

    # ---- variables removed by alias elimination, recovered for output ----
    src__i = -c__i    # src.i
    src__p__i = -c__i    # src.p.i
    src__n__i = c__i    # src.n.i
    r__i = c__i    # r.i
    r__p__i = c__i    # r.p.i
    r__n__i = -c__i    # r.n.i
    c__p__i = c__i    # c.p.i
    c__n__i = -c__i    # c.n.i
    src__p__v = src__V    # src.p.v
    src__n__v = 0.0    # src.n.v
    c__n__v = 0.0    # c.n.v
    gnd__p__v = 0.0    # gnd.p.v
    r__n__v = c__v    # r.n.v
    src__v = src__V    # src.v
    c__p__v = c__v    # c.p.v
    r__p__v = src__V    # r.p.v

    # ---- results ----
    der = np.array([der_c__v])
    variables = {
        'r.v': r__v,
        'c.v': c__v,
        'c.i': c__i,
        'gnd.p.i': gnd__p__i,
        'src.i': src__i,
        'src.p.i': src__p__i,
        'src.n.i': src__n__i,
        'r.i': r__i,
        'r.p.i': r__p__i,
        'r.n.i': r__n__i,
        'c.p.i': c__p__i,
        'c.n.i': c__n__i,
        'src.p.v': src__p__v,
        'src.n.v': src__n__v,
        'c.n.v': c__n__v,
        'gnd.p.v': gnd__p__v,
        'r.n.v': r__n__v,
        'src.v': src__v,
        'c.p.v': c__p__v,
        'r.p.v': r__p__v,
        'der(c.v)': der_c__v,
    }
    events = []
    return {'der': der, 'variables': variables, 'events': events}

How this system is actually solved

The blocks above are the plan. This is the procedure that follows from it: what happens to each block, what happens before the first step, and what happens when an event interrupts.

1. Once, before the first step: initialization

Every state simply takes its start value; no equations have to be solved yet:

c.v = 0

2. At every evaluation: the continuous system

The integrator supplies the states c.v, the parameters, time, and asks what the derivatives are. Everything else is unknown and has to be computed, in the order the sorting found:

  1. solve simultaneously for r1.v, r1.i, r2.i, r2.v, r3.v, c.i:
    r1.v = src.V - r2.v
    r1.v = r1.R * r1.i
    r2.v = r2.R * r2.i
    r3.v = r2.v - c.v
    r3.v = r3.R * c.i
    -r1.i + r2.i + c.i = 0
    These 6 equations depend on each other in a circle, so none of them can be solved on its own. They are linear in those unknowns, so the block is one matrix solve A x = b: direct, and exact up to rounding.
  2. der(c.v) := c.i/c.Crearranged directly from c.C * der(c.v) = c.i
  3. gnd.p.i := c.i + r2.i - r1.irearranged directly from r1.i - r2.i - c.i + gnd.p.i = 0

The derivatives der(c.v) go back to the integrator. Every other value computed on the way is kept as a result, and the variables that alias elimination removed are recovered from the ones that survived.

3. In between: the integrator

Between evaluations the integrator advances the states -- either choosing its own step size to meet a tolerance, or taking the fixed step you gave it. That is the only part of this whole pipeline that TinySim does not do itself.

# Simulation code generated by TinySim for model 'ResistorNetwork'.
#

# States: c.v

# Dots in model names become double underscores here: c.v -> c__v.

from math import (sin, cos, tan, asin, acos, atan, atan2, exp, log,
                  log10, sqrt, tanh, fabs as abs)
import numpy as np
from scipy.optimize import fsolve


def _solve_block(residual, guess, block, unknowns, t):
    """
    Solve one algebraic block numerically.

    An iterative solver needs a starting point, and a bad one can
    make it fail on a system that has a perfectly good solution --
    which is exactly why `start` values matter for the variables of
    an algebraic loop.  This tries the remembered guess first, then
    a fresh start from zero, and refuses to continue quietly if
    neither works.
    """
    for attempt in (guess, [0.0] * len(guess)):
        values, _, flag, message = fsolve(residual, attempt,
                                          full_output=True)
        if flag == 1:
            return values
    raise RuntimeError(
        'block %d (%s) did not converge at t = %g: %s'
        % (block, unknowns, t, message.strip()))


def evaluate(t, x, p, d, guess=None):
    """

    c__v = x[0]    # c.v

    r1__R = p['r1.R']    # r1.R -- resistance [Ohm]
    r2__R = p['r2.R']    # r2.R -- resistance [Ohm]
    r3__R = p['r3.R']    # r3.R -- resistance [Ohm]
    src__V = p['src.V']    # src.V

    # ---- block 1: algebraic loop of size 6: solve for r1.v, r1.i, r2.i, r2.v, r3.v, c.i  [linear system]
    #        r1.v = src.V - r2.v
    #        r1.v = r1.R * r1.i
    #        r2.v = r2.R * r2.i
    #        r3.v = r2.v - c.v
    #        r3.v = r3.R * c.i
    #        -r1.i + r2.i + c.i = 0

    c__n__v = 0.0    # c.n.v

    src__v = src__V    # src.v

    r1__p__v = src__V    # r1.p.v

    # ---- results ----
