Nothing in the tool knows about electricity or mechanics. Both connectors declare one potential variable and one flow variable, and the same two rules apply to both: potentials in a connection set are equal, flows sum to zero. On the electrical side that is Kirchhoff; on the mechanical side it is the balance of torques. The flat equations below are the proof.

The model -- dcmotor.tiny

// Two physical domains joined by one component: the classic DC motor.
// Shows that the SAME connect() rules give Kirchhoff's laws on the electrical
// side and Newton's third law on the mechanical side.

connector Pin
  Real v;
  flow Real i;
end Pin;

connector Flange
  Real phi      "angle [rad]";
  flow Real tau "torque INTO the component [N.m]";
end Flange;

partial model OnePort
  Pin p, n;
  Real v, i;
equation
  v = p.v - n.v;
  i = p.i;
  p.i + n.i = 0;
end OnePort;

model Resistor
  extends OnePort;
  parameter Real R = 0.5;
equation
  v = R * i;
end Resistor;

model Inductor
  extends OnePort;
  parameter Real L = 0.05;
equation
  L * der(i) = v;
end Inductor;

model ConstantVoltage
  extends OnePort;
  parameter Real V = 24;
equation
  v = V;
end ConstantVoltage;

model Ground
  Pin p;
equation
  p.v = 0;
end Ground;

model EMF "electro-mechanical energy conversion"
  extends OnePort;
  parameter Real k = 0.1 "torque constant [N.m/A]";
  Flange flange;
  Real w "shaft speed [rad/s]";
equation
  w = der(flange.phi);
  v = k * w;              // back-EMF
  flange.tau = -k * i;    // torque delivered to the shaft
end EMF;

model Inertia
  parameter Real J = 0.02 "moment of inertia [kg.m2]";
  parameter Real d = 0.01 "viscous friction [N.m.s]";
  Flange flange;
  Real phi(start = 0), w(start = 0);
equation
  phi = flange.phi;
  der(phi) = w;
  J * der(w) = flange.tau - d * w;
end Inertia;

model DCMotor
  ConstantVoltage src(V = 24);
  Resistor r(R = 0.5);
  Inductor l(L = 0.05, i(start = 0));
  EMF emf(k = 0.1);
  Inertia load(J = 0.02, d = 0.01);
  Ground gnd;
equation
  connect(src.p, r.p);
  connect(r.n, l.p);
  connect(l.n, emf.p);
  connect(emf.n, src.n);
  connect(src.n, gnd.p);
  connect(emf.flange, load.flange);
end DCMotor;

# ==========================================================================
# Simulation code generated by TinySim for model 'DCMotor'.
#
# 10 equations, 10 unknowns, sorted into 10 blocks (no algebraic loops)
# States: l.i, load.phi, load.w
#
# Dots in model names become double underscores here: c.v -> c__v.
# ==========================================================================

from math import (sin, cos, tan, asin, acos, atan, atan2, exp, log,
                  log10, sqrt, tanh, fabs as abs)
import numpy as np
from scipy.optimize import fsolve


def _solve_block(residual, guess, block, unknowns, t):
    """
    Solve one algebraic block numerically.

    An iterative solver needs a starting point, and a bad one can
    make it fail on a system that has a perfectly good solution --
    which is exactly why `start` values matter for the variables of
    an algebraic loop.  This tries the remembered guess first, then
    a fresh start from zero, and refuses to continue quietly if
    neither works.
    """
    for attempt in (guess, [0.0] * len(guess)):
        values, _, flag, message = fsolve(residual, attempt,
                                          full_output=True)
        if flag == 1:
            return values
    raise RuntimeError(
        'block %d (%s) did not converge at t = %g: %s'
        % (block, unknowns, t, message.strip()))


def evaluate(t, x, p, d, guess=None):
    """
    Solve the simulation equations of model 'DCMotor'.

    t     : time
    x     : state vector, in the order listed below
    p     : dict of parameter values, keyed by model name
    d     : dict of discrete-variable values
    guess : dict used to remember iteration start values
    """
    guess = {} if guess is None else guess

    # ---- states: supplied by the integrator ----
    l__i = x[0]    # l.i
    load__phi = x[1]    # load.phi
    load__w = x[2]    # load.w

    # ---- parameters: constant during simulation ----
    emf__k = p['emf.k']    # emf.k -- torque constant [N.m/A]
    l__L = p['l.L']    # l.L
    load__J = p['load.J']    # load.J -- moment of inertia [kg.m2]
    load__d = p['load.d']    # load.d -- viscous friction [N.m.s]
    r__R = p['r.R']    # r.R
    src__V = p['src.V']    # src.V

    # ---- block 1: solve for r.v  [explicit]
    #        r.v = r.R * l.i
    r__v = l__i*r__R

    # ---- block 2: solve for l.p.v  [explicit]
    #        r.v = src.V - l.p.v
    l__p__v = -r__v + src__V

    # ---- block 3: solve for der(load.phi)  [explicit]
    #        der(load.phi) = load.w
    der_load__phi = load__w

    # ---- block 4: solve for emf.w  [explicit]
    #        emf.w = der(load.phi)
    emf__w = der_load__phi

    # ---- block 5: solve for emf.v  [explicit]
    #        emf.v = emf.k * emf.w
    emf__v = emf__k*emf__w

    # ---- block 6: solve for l.v  [explicit]
    #        l.v = l.p.v - emf.v
    l__v = -emf__v + l__p__v

    # ---- block 7: solve for der(l.i)  [explicit]
    #        l.L * der(l.i) = l.v
    der_l__i = l__v/l__L

    # ---- block 8: solve for emf.flange.tau  [explicit]
    #        emf.flange.tau = -(emf.k * l.i)
    emf__flange__tau = -emf__k*l__i

    # ---- block 9: solve for der(load.w)  [explicit]
    #        load.J * der(load.w) = -emf.flange.tau - load.d * load.w
    der_load__w = (-emf__flange__tau - load__d*load__w)/load__J

    # ---- block 10: solve for gnd.p.i  [explicit]
    #        l.i - l.i + gnd.p.i = 0
    gnd__p__i = 0

    src__p__i = -l__i    # src.p.i
    src__n__i = l__i    # src.n.i
    r__i = l__i    # r.i
    r__p__i = l__i    # r.p.i
    r__n__i = -l__i    # r.n.i
    l__p__i = l__i    # l.p.i
    l__n__i = -l__i    # l.n.i
    emf__i = l__i    # emf.i
    emf__p__i = l__i    # emf.p.i
    emf__n__i = -l__i    # emf.n.i
    load__flange__phi = load__phi    # load.flange.phi
    emf__flange__phi = load__phi    # emf.flange.phi
    src__p__v = src__V    # src.p.v
    src__n__v = 0.0    # src.n.v
    emf__n__v = 0.0    # emf.n.v
    gnd__p__v = 0.0    # gnd.p.v
    r__n__v = l__p__v    # r.n.v
    emf__p__v = emf__v    # emf.p.v
    load__flange__tau = -emf__flange__tau    # load.flange.tau
    src__v = src__V    # src.v
    l__n__v = emf__v    # l.n.v
    r__p__v = src__V    # r.p.v

    # ---- results ----
    der = np.array([der_l__i, der_load__phi, der_load__w])
    variables = {
        'r.v': r__v,

"""
Experiment 6 -- one connect rule, two physical domains.

    python experiments/06_dc_motor.py

The DC motor joins an electrical circuit to a rotating shaft.  Nothing in the
tool knows about electricity or mechanics: the same two rules -- potentials
equal, flows sum to zero -- give Kirchhoff's current law on one side and the
balance of torques on the other.  The proof is in the flat equations.

    python experiments/06_dc_motor.py --html
"""

import pathlib

import matplotlib.pyplot as plt

import setup_path  # noqa: F401  (lets this run without installing TinySim)
import tinysim
from tinysim import htmlreport

EXAMPLES = pathlib.Path(__file__).resolve().parent.parent / "examples"
FIGURES = pathlib.Path(__file__).resolve().parent.parent / "figures"
FIGURES.mkdir(exist_ok=True)

page = htmlreport.start(
    __file__,
    title="Experiment 6 - one connect rule, two physical domains",
    subtitle="A DC motor: Kirchhoff's laws and the balance of torques from the same two lines")

motor = tinysim.load(EXAMPLES / "dcmotor.tiny", "DCMotor")
tinysim.explain(motor, "connections")

page.add_text("""
    Nothing in the tool knows about electricity or mechanics. Both connectors
    declare one potential variable and one flow variable, and the same two
    rules apply to both: potentials in a connection set are equal, flows sum to
    zero. On the electrical side that is Kirchhoff; on the mechanical side it
    is the balance of torques. The flat equations below are the proof.
    """)
page.add_source(EXAMPLES / "dcmotor.tiny")
page.add_model(motor)

print("\nequations that came from connect():")
for equation in motor.flat.equations:
    if equation.origin.startswith("connect"):
        print(f"    {equation.source:<40} # {equation.origin}")

result = tinysim.simulate(motor, stop=2.0, points=2001)

voltage, resistance, k, damping = 24.0, 0.5, 0.1, 0.01
steady_state = voltage * k / (k * k + resistance * damping)
print(f"\nfinal speed {result['load.w'][-1]:.2f} rad/s, "
      f"steady state from hand calculation {steady_state:.2f} rad/s")

figure, (speed, current) = plt.subplots(2, 1, sharex=True, figsize=(8, 5))
speed.plot(result.time, result["load.w"])
speed.axhline(steady_state, color="grey", linestyle=":", label="steady state")
speed.set_ylabel("shaft speed [rad/s]")
speed.set_title("DC motor starting up")
speed.legend()
speed.grid(alpha=0.3)

current.plot(result.time, result["l.i"], color="tab:red")
current.set_ylabel("armature current [A]")

current.grid(alpha=0.3)
figure.tight_layout()
figure.savefig(FIGURES / "dc_motor.png", dpi=150)
print(f"wrote {FIGURES / 'dc_motor.png'}")
page.add_result(result, ["load.w", "load.phi", "l.i", "emf.v"],
                title="The simulation - starting up")
page.add_figure(figure, f"The shaft settles at the speed a hand calculation "
                        f"gives, {steady_state:.1f} rad/s.")
page.finish()
